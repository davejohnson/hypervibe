  async function authenticateCredentialRequests(token) {
    const result = await authenticate(token, PERMISSIONS.CREDENTIALS_MANAGE);
    const membership = await result.credential.getCreatedByMembership();
    const role = membership && await membership.getRole();
    if (!result.credential.expiresAt || !membership || membership.status !== "active" || membership.revokedAt
      || membership.AccountId !== result.account.id || !role || role.AccountId !== result.account.id
      || !hasPermission({ role }, PERMISSIONS.CREDENTIALS_MANAGE)) throw unauthorized("Credential-request access expired or was revoked. Approve this device again.");
    return { account: result.account, membership, role, credentialProjectId: result.project.id, credentialEnvironmentId: result.environment.id };
  }
  async function revokeCredentialRequestAccess(token) {
    await authenticateCredentialRequests(token);
    const { credential } = await authenticate(token, PERMISSIONS.CREDENTIALS_MANAGE);
    await credential.update({ revokedAt: new Date() });
    return { applied: 1, skipped: 0 };
  }
