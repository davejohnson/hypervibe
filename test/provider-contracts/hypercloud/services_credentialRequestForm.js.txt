const { invalid, unprocessable } = require("../lib/serviceError");

function emailAddress(value) {
  const email = typeof value === "string" ? value.trim().toLowerCase() : "";
  if (email.length > 320 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || /[\u0000-\u001f\u007f]/.test(email)) throw invalid("Enter the credential owner’s email address.");
  return email;
}

function fieldsForSource(source, environment) {
  return Object.entries(source.spec.secrets || {}).filter(([, slot]) =>
    slot && (slot.ownership || "delegated") === "delegated" && slot.environments?.includes(environment.key)
  ).map(([key]) => ({ key, label: key.replaceAll("_", " "), inputType: "password" }));
}

function requestFields(input, available) {
  if (!Array.isArray(input) || !input.length || input.length > 16) throw invalid("Select between 1 and 16 credential fields.");
  const names = new Set(available.map(({ key }) => key));
  const used = new Set();
  return input.map((field) => {
    if (!field || Object.keys(field).some((key) => !["key", "label", "inputType"].includes(key)) ||
      !names.has(field.key) || used.has(field.key)) throw unprocessable("Select each declared credential once. Generated application secrets cannot be requested.");
    const label = typeof field.label === "string" ? field.label.trim() : field.key.replaceAll("_", " ");
    const inputType = field.inputType || "password";
    if (!label || label.length > 120 || /[\u0000-\u001f\u007f]/.test(label) || !["password", "text", "textarea"].includes(inputType)) throw invalid("Use a short field label and a supported field format.");
    used.add(field.key);
    return { key: field.key, label, inputType };
  });
}

module.exports = { emailAddress, fieldsForSource, requestFields };
