  async function create(request, response) {
    const keys = Array.isArray(request.body.keys) ? request.body.keys : request.body.keys ? [request.body.keys] : [];
    const input = Object.hasOwn(request.body, "recipientEmail") ? {
      expectedSourceRevision: request.body.expectedSourceRevision, expectedSourceDigest: request.body.expectedSourceDigest,
      recipientEmail: request.body.recipientEmail, title: request.body.title, sourceBranch: request.body.sourceBranch,
      fields: request.path.startsWith("/api/") ? request.body.fields : keys.map((key) => ({ key, label: request.body[`label.${key}`], inputType: request.body[`format.${key}`] }))
    } : { recipient: request.body.recipient, sourceBranch: request.body.sourceBranch };
    const result = await service().create({
      ...ownerInput(request),
      environmentId: request.body.environmentId,
      input
    });
    if (request.path.startsWith("/api/"))
      return response.status(201).json(result);
    return renderOwner(request, response, { mode: "created", ...result }, 201);
