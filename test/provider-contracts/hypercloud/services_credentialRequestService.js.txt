function requestView(record, now, environment = record.environment) {
  return {
    id: record.id,
    projectId: record.ProjectId,
    environmentId: record.EnvironmentId,
    recipient: record.recipientEmail || record.recipientLogin,
    authentication: record.recipientEmail ? "email" : "github",
    title: record.title || "Deployment credentials",
    fields: record.fields || record.keys.map((key) => ({ key, label: key, inputType: "password" })),
    delivery: (record.emails || []).map((email) => ({ id: email.id, kind: email.kind, status: email.status, failureCode: email.failureCode })),
    sourceBranch: record.sourceBranch || null,
    sourceRevision: record.sourceRevision,
    environment: environment
      ? {
          id: environment.id,
          key: environment.key,
          name: environment.name
        }
      : null,
    status:
      ["pending", "ready"].includes(record.status) && record.expiresAt <= now
        ? "expired"
        : record.status,
    keys: record.keys,
    suppliedKeys: record.suppliedKeys,
    submissions: record.submissions,
    expiresAt: record.expiresAt.toISOString(),
    submittedAt: record.submittedAt?.toISOString() || null,
    consumedAt: record.consumedAt?.toISOString() || null,
    verification: "presence_only",
    deployment: "not_authorized"
  };
}

  async function createEmailRequest({ actor, projectId, environmentId, input }) {
    if (!deliveryService?.enqueue) throw unavailable("Credential email delivery is not configured.");
    if (Object.keys(input).some((key) => !["recipientEmail", "title", "fields", "sourceBranch"].includes(key))) throw invalid("Provide an owner, request name, and credential fields.");
    const recipientEmail = emailAddress(input.recipientEmail);
    const title = typeof input.title === "string" ? input.title.trim() : "Deployment credentials";
    if (!title || title.length > 120 || /[\u0000-\u001f\u007f]/.test(title)) throw invalid("Give this credential request a short name.");
    let sourceBranch;
    try { sourceBranch = normalizeSourceBranch(input.sourceBranch); }
    catch { throw invalid("Enter a branch name, not a tag, revision, or full ref."); }
    const scope = await scopeForOwner(actor, projectId, environmentId);
    if (!scope.environment) throw invalid("Select an environment.");
    const source = await sourceFor(scope, sourceBranch);
    const fields = requestFields(input.fields, fieldsForSource(source, scope.environment));
    const keys = fields.map(({ key }) => key).sort();
    const record = await database.transaction(async (transaction) => {
      await scope.environment.reload({ transaction, lock: transaction.LOCK.UPDATE });
      const live = { status: { [Op.in]: ["pending", "ready"] }, expiresAt: { [Op.gt]: clock() } };
      if (await scope.environment.countCredentialRequests({ where: live, transaction }) >= 20) throw conflict("Revoke an existing request before creating another.");
      await ensureUnassigned(scope.environment, keys, transaction);
      const request = await scope.account.createCredentialRequest({
        ProjectId: scope.project.id, EnvironmentId: scope.environment.id, CreatedByMembershipId: actor.membership.id,
        recipientEmail, title, fields, keys, sourceBranch, sourceRevision: source.revision, sourceDigest: source.digest,
        invitationTokenHash: hashOpaqueToken(generateOpaqueToken()), expiresAt: new Date(clock().getTime() + REQUEST_LIFETIME_MS)
      }, { transaction });
      await deliveryService.enqueue({ request, kind: "invitation", transaction });
      return request;
    });
    return { applied: 1, skipped: 0, request: view(record, clock(), scope.environment), delivery: "queued",
      invitationUrl: publicResourceUrl(publicBaseUrl, `credential-requests/${record.id}`) };
  }

  async function prepare({ actor, projectId, environmentId, sourceBranch }) {
    permission(actor, PERMISSIONS.CREDENTIALS_MANAGE);
    const scope = await scopeForOwner(actor, projectId);
    const environments = await scope.project.getEnvironments({
      limit: 100,
      order: [
        ["name", "ASC"],
        ["id", "ASC"]
      ]
    });
    const selected = environmentId ? environments.find(({ id }) => id === environmentId)
      : environments.find(({ key }) => key === "staging") || environments[0];
    if (environmentId && !selected) throw notFound();
    let availableFields = [];
    let sourceError = null;
    if (selected) {
      let branch;
      try { branch = normalizeSourceBranch(sourceBranch); }
      catch { throw invalid("Enter a branch name, not a tag, revision, or full ref."); }
      try {
        availableFields = fieldsForSource(await sourceFor({ ...scope, environment: selected }, branch), selected);
      } catch (error) {
        if (!error.expose || ![409, 503].includes(error.status)) throw error;
        sourceError = { code: error.code, message: error.message };
      }
    }
    return {
      availableFields, sourceError, selectedEnvironmentId: selected?.id || null, sourceBranch: sourceBranch || "",
      environments: environments.map((environment) => ({
        id: environment.id,
        name: environment.name,
        preferred: environment.key === "staging"
      }))
    };
  }

  async function list({ actor, projectId, page = 1, sort = "newest" }) {
    permission(actor, PERMISSIONS.OPERATIONS_READ);
    const scope = await scopeForOwner(actor, projectId);
    const selectedPage = Math.min(
      1000,
      Math.max(1, Number.parseInt(page, 10) || 1)
    );
    const total = await scope.project.countCredentialRequests();
    const direction = sort === "oldest" ? "ASC" : "DESC";
    const records = await scope.project.getCredentialRequests({
      include: [
        { association: "environment", attributes: ["id", "key", "name"] },
        { association: "emails", separate: true, limit: 2, order: [["createdAt", "DESC"]] }
      ],
      limit: 20,
      offset: (selectedPage - 1) * 20,
      order: [
        ["createdAt", direction],
        ["id", direction]
      ]
    });
    return {
      requests: records.map((record) => view(record, clock())),
      pagination: {
        page: selectedPage,
        pageSize: 20,
        total,
        pageCount: Math.ceil(total / 20),
        hasNext: selectedPage * 20 < total,
        hasPrevious: selectedPage > 1
      }
    };
  }

  async function revoke(input) {
    const { scope, record } = await ownerRecord(
      input,
      PERMISSIONS.CREDENTIALS_MANAGE
    );
    return database.transaction(async (transaction) => {
      const [locked] = await scope.project.getCredentialRequests({
        where: { id: record.id },
        transaction,
        lock: transaction.LOCK.UPDATE
      });
      if (!locked) throw notFound();
      if (["revoked", "consumed"].includes(locked.status))
        return {
          applied: 0,
          skipped: 1,
          request: view(locked, clock(), record.environment)
        };
      await locked.update(
        { status: "revoked", revokedAt: clock(), credentialEnvelope: null, emailCodeHash: null, emailProofHash: null, emailCodeExpiresAt: null },
        { transaction }
      );
      return {
        applied: 1,
        skipped: 0,
        request: view(locked, clock(), record.environment)
      };
    });
  }
