const { NotifierError } = require("../lib/notifier");
const { unavailable } = require("../lib/serviceError");
const { hasPermission, PERMISSIONS } = require("./permissions");
const WORKFLOW_TYPE = "credential_request_email";
const UUID = /^[a-f0-9]{8}-[a-f0-9]{4}-[1-5][a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/i;

// Incident outbox rows require an incident and verified alert destination.
// Credential correspondence uses its own small receipt ledger, sharing the
// existing notification transport/worker and durable queue implementation.
function createCredentialRequestDeliveryService({ database, notifier, workflowQueue, clock = () => new Date() }) {
  async function enqueue({ request, kind, transaction }) {
    if (!notifier?.sendCredentialRequest || !workflowQueue) throw unavailable("Credential email delivery is not configured.");
    if (!["invitation", "ready"].includes(kind)) throw new TypeError("Invalid credential email kind.");
    const [existing] = await request.getEmails({ where: { kind }, limit: 1, transaction });
    if (existing) return { applied: 0, skipped: 1, status: existing.status };
    const email = await request.createEmail({ AccountId: request.AccountId, kind, status: "queued" }, { transaction });
    await workflowQueue.enqueue({ type: WORKFLOW_TYPE, version: 1, emailId: email.id }, {
      transaction, idempotencyKey: `credential-email:${email.id}`
    });
    return { applied: 1, skipped: 0, status: "queued" };
  }

  async function sendVerification({ to, code }) {
    if (!notifier?.sendCredentialCode) throw unavailable("Credential email verification is not configured.");
    try { await notifier.sendCredentialCode({ to, code }); }
    catch { throw unavailable("The verification email could not be sent. Wait a minute and request another code."); }
  }

  async function processWorkflow({ data }) {
    if (data?.type !== WORKFLOW_TYPE || data.version !== 1 || !UUID.test(data.emailId || "") ||
      Object.keys(data).some((key) => !["type", "version", "emailId"].includes(key))) throw new TypeError("Invalid credential email workflow.");
    const claimed = await database.transaction(async (transaction) => {
      // OK: This UUID is trusted internal queue state; re-resolve the request through its account before delivery.
      const email = await database.models.CredentialRequestEmail.findOne({ where: { id: data.emailId }, transaction, lock: transaction.LOCK.UPDATE });
      if (!email || ["sent", "failed", "unknown", "suppressed"].includes(email.status)) return null;
      if (email.status === "retrying" && email.nextAttemptAt > clock()) return null;
      if (email.status === "sending") {
        if (clock() - email.startedAt > 60_000) await email.update({ status: "unknown", failureCode: "delivery_outcome_unknown" }, { transaction });
        return null;
      }
      const account = await email.getAccount({ transaction });
      const [request] = account ? await account.getCredentialRequests({ where: { id: email.CredentialRequestId }, limit: 1, transaction }) : [];
      const project = request && await request.getProject({ transaction });
      const environment = request && await request.getEnvironment({ transaction });
      const membership = request && await request.getCreatedByMembership({ transaction });
      const role = membership && await membership.getRole({ transaction });
      const user = membership && await membership.getUser({ transaction });
      if (!request || !project || !environment || !request.recipientEmail || !["pending", "ready"].includes(request.status) ||
        request.expiresAt <= clock() || !membership || membership.status !== "active" || membership.revokedAt ||
        !role || !hasPermission({ role }, PERMISSIONS.CREDENTIALS_MANAGE) || !user ||
        (email.kind === "ready" && request.status !== "ready")) {
        await email.update({ status: "suppressed", failureCode: "request_unavailable" }, { transaction });
        return null;
      }
      await email.update({ status: "sending", startedAt: clock(), nextAttemptAt: null, attemptCount: email.attemptCount + 1 }, { transaction });
      return { email, input: { to: email.kind === "invitation" ? request.recipientEmail : user.email,
        requestId: request.id, projectId: project.id, projectName: project.name, title: request.title, kind: email.kind } };
    });
    if (!claimed) return { applied: 0, skipped: 1 };
    let accepted = false;
    try {
      if (!notifier?.sendCredentialRequest) throw new NotifierError({ code: "provider_not_configured", message: "Credential delivery is unavailable." });
      const receipt = await notifier.sendCredentialRequest(claimed.input);
      accepted = true;
      await claimed.email.update({ status: "sent", deliveredAt: clock(), receiptId: receipt.receiptId, failureCode: null });
      return { applied: 1, skipped: 0, outcome: "sent" };
    } catch (error) {
      const unknown = accepted || !(error instanceof NotifierError) || ["provider_timeout_unknown", "provider_receipt_missing"].includes(error.code);
      const retry = !unknown && error.retryable && claimed.email.attemptCount < 4;
      await database.transaction(async (transaction) => {
        const nextAttemptAt = retry ? new Date(clock().getTime() + Math.max(error.retryAfterMs || 0, 60_000 * 2 ** (claimed.email.attemptCount - 1))) : null;
        await claimed.email.update({ status: retry ? "retrying" : unknown ? "unknown" : "failed", nextAttemptAt,
          failureCode: unknown ? "delivery_outcome_unknown" : error.code }, { transaction });
        if (retry) await workflowQueue.enqueue(data, { transaction, startAfter: nextAttemptAt,
          idempotencyKey: `credential-email:${claimed.email.id}:attempt:${claimed.email.attemptCount + 1}` });
      });
      return { applied: 1, skipped: 0, outcome: retry ? "retrying" : unknown ? "unknown" : "failed" };
    }
  }
  return { enqueue, processWorkflow, sendVerification };
}

module.exports = { createCredentialRequestDeliveryService, WORKFLOW_TYPE };
